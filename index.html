// ========================================== // SPLASH INTRO CONTROLLER (Smooth & Lightweight) // ========================================== const splashSlides = [ 'splashSlide1', 'splashSlide2', 'splashSlide3', 'splashSlide4', 'splashSlide5', 'splashSlide6', 'splashSlide7' ]; let currentSplashIdx = 0; let splashTimer = null; function runSplashScreen() { const splashEl = document.getElementById('splashScreen'); if (!splashEl) return; const totalSlides = splashSlides.length; const intervalTime = 1600; // प्रत्येक स्लाइड १.६ सेकंद दिसेल splashTimer = setInterval(() => { // जुनी स्लाइड लपवा const prevSlide = document.getElementById(splashSlides[currentSplashIdx]); if (prevSlide) prevSlide.classList.remove('active'); currentSplashIdx++; // प्रोग्रेस बार अपडेट करा const progress = (currentSplashIdx / (totalSlides - 1)) * 100; const progBar = document.getElementById('splashProgress'); if (progBar) progBar.style.width = `${progress}%`; if (currentSplashIdx < totalSlides) { const nextSlide = document.getElementById(splashSlides[currentSplashIdx]); if (nextSlide) nextSlide.classList.add('active'); } else { // सर्व स्लाइड्स संपल्या, स्प्लॅश बंद करा clearInterval(splashTimer); setTimeout(dismissSplash, 1200); } }, intervalTime); } function dismissSplash() { if (splashTimer) clearInterval(splashTimer); const splashEl = document.getElementById('splashScreen'); if (splashEl) { splashEl.style.opacity = '0'; setTimeout(() => { splashEl.style.display = 'none'; }, 700); } } // पेज लोड झाल्यावर लगेच स्प्लॅश सुरू करा document.addEventListener('DOMContentLoaded', () => { runSplashScreen(); });
